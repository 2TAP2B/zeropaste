zeropaste - zero-knowledge pastes · · source code · dev
self-host in 30 seconds 
docker run -d --name zeropaste -p 8080:8080 -v paste-data:/data ghcr.io/2tap2b/zeropaste:latest
or compose.yaml - edit the values, then start
services:
  paste:
    image: ghcr.io/2tap2b/zeropaste:latest
    ports:
      - "8080:8080"
    volumes:
      - paste-data:/data
    restart: unless-stopped
    environment:
      CREATE_KEY: "change-me"          # site passphrase in front of creates; delete the line for a fully public instance
      USER_ACCOUNTS: "true"            # "false" = no accounts at all
      USER_REGISTRATION: "true"        # "false" = sign-up closes after the first account
      MAX_TTL_DAYS: "30"               # share-time ceiling in days
      MAX_SHARES_PER_ACCOUNT: "0"      # active shares per account; 0 = unlimited
      MAX_UPLOAD: "104857600"          # bytes per share; default 100 MiB
      REG_OPEN: "false"                # static registration lock
      CHUNK: "8388608"                 # bytes per upload chunk
volumes:
  paste-data:
