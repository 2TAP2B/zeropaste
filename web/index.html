self-host in 30 seconds 
docker run -d --name zeropaste -p 8080:8080 -v paste-data:/data ghcr.io/2tap2b/zeropaste:latest
or compose.yaml
services:
  paste:
    image: ghcr.io/2tap2b/zeropaste:latest
    ports:
      - "8080:8080"
    environment:
      CREATE_KEY: "change-me" # site passphrase for creating; delete line for fully public
    volumes:
      - paste-data:/data
    restart: unless-stopped
volumes:
  paste-data:
Plain HTTP; put your TLS reverse proxy in front. Open the address and share — links assemble client-side.
